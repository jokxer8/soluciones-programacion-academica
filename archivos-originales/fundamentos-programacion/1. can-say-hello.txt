using System;
class Program
{
    static bool CanSayHello(string s)
    {
        int index = 0;
        string hello = "hello";
        foreach (char c in s)
        {
            if (c == hello[index])
            {
                index++;
            }
            if (index == hello.Length)
            {
                return true;
            }
        }
        return false;
    }
    static void Main(string[] args)
    { 
        string s = Console.ReadLine();
        if (CanSayHello(s))
        { 
            Console.WriteLine("YES");
        } 
        else
        { 
            Console.WriteLine("NO");
        }
        Console.ReadLine();
    }
}
