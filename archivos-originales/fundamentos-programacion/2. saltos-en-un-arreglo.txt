using System;

class Program
{
    static void Main()
    {
        int[] A = { 1, 3, 5, 8, 9, 2, 6, 7, 6, 8, 9 };

        int result = MinJumpsToEnd(A);

        if (result != -10)
        {
            Console.WriteLine($"El número mínimo de saltos para llegar al final es: {result}");
        }
        else
        {
            Console.WriteLine("No es posible llegar al final desde el índice 0.");
        }
    }

    static int MinJumpsToEnd(int[] arr)
    {
        int n = arr.Length;

        if (arr[0] == 0)
        {
            return -10;
        }

        int[] jumps = new int[n];
        jumps[0] = 0;

        for (int i = 1; i < n; i++)
        {
            jumps[i] = int.MaxValue;

            for (int j = 0; j < i; j++)
            {
                if (j + arr[j] >= i && jumps[j] != int.MaxValue)
                {
                    jumps[i] = Math.Min(jumps[i], jumps[j] + 1);
                    break;
                }
            }
        }

        if (jumps[n - 1] != int.MaxValue)
        {
            return jumps[n - 1];
        }
        else
        {
            return -10;
        }
    }
}
