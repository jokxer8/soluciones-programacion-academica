using System;

class Program
{
    static void Main()
    {
        int n = LeerEntero("Ingrese la cantidad de nodos: ");
        int m = LeerEntero("Ingrese la cantidad de aristas: ");
        int[,] grafo = new int[n, n];

        Console.WriteLine("Ingrese cada arista en formato: origen destino peso");
        for (int i = 0; i < m; i++)
        {
            int[] datos = LeerTresEnteros($"Arista {i + 1}: ");
            int u = datos[0];
            int v = datos[1];
            int peso = datos[2];

            if (grafo[u, v] == 0 || peso < grafo[u, v])
            {
                grafo[u, v] = peso;
                grafo[v, u] = peso;
            }
        }

        Prim(grafo);
    }

    static void Prim(int[,] grafo)
    {
        int n = grafo.GetLength(0);
        bool[] enArbol = new bool[n];
        int[] clave = new int[n];
        int[] padre = new int[n];
        for (int i = 0; i < n; i++)
        {
            clave[i] = int.MaxValue;
            padre[i] = -1;
        }
        clave[0] = 0;
        for (int i = 0; i < n; i++)
        {
            int u = BuscarMinimo(clave, enArbol);
            if (u == -1)
            {
                Console.WriteLine("No se puede formar un árbol de expansión mínima porque el grafo no es conexo.");
                return;
            }
            enArbol[u] = true;
            for (int v = 0; v < n; v++)
            {
                if (grafo[u, v] != 0 && !enArbol[v] && grafo[u, v] < clave[v])
                {
                    clave[v] = grafo[u, v];
                    padre[v] = u;
                }
            }
        }
        int total = 0;
        Console.WriteLine("\nAristas del árbol de expansión mínima:");

        for (int i = 1; i < n; i++)
        {
            if (padre[i] == -1)
            {
                Console.WriteLine("No se puede formar un árbol de expansión mínima porque el grafo no es conexo.");
                return;
            }
            Console.WriteLine($"{padre[i]} - {i}   peso: {grafo[padre[i], i]}");
            total += grafo[padre[i], i];
        }
        Console.WriteLine("Peso total: " + total);
    }

    static int BuscarMinimo(int[] clave, bool[] enArbol)
    {
        int minimo = int.MaxValue;
        int indice = -1;
        for (int i = 0; i < clave.Length; i++)
        {
            if (!enArbol[i] && clave[i] < minimo)
            {
                minimo = clave[i];
                indice = i;
            }
        }
        return indice;
    }

    static int LeerEntero(string mensaje)
    {
        Console.Write(mensaje);
        return int.Parse(Console.ReadLine());
    }

    static int[] LeerTresEnteros(string mensaje)
    {
        Console.Write(mensaje);
        string[] partes = Console.ReadLine().Split(new char[] { ' ' }, StringSplitOptions.RemoveEmptyEntries);
        return new int[] {
            int.Parse(partes[0]),
            int.Parse(partes[1]),
            int.Parse(partes[2])
        };
    }
}
