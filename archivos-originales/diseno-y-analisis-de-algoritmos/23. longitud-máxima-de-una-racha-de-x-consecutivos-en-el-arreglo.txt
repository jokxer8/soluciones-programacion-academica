Longitud máxima de una racha de x consecutivos en el arreglo

A) Recursivo con DP + entrada recursiva

using System;
using System.Collections.Generic;

class RachaMaximaX_RecursivoDP
{
    static void Main()
    {
        int n = LeerEnteroRec("Cantidad N: ");
        int[] arreglo = new int[n];
        LeerArregloRec(arreglo, 0);

        int x = LeerEnteroRec("Número x: ");

        var memo = new Dictionary<int, (int mejorDesdeAqui, int rachaDesdeAqui)>();
        int mejor = MejorRachaRecDP(arreglo, x, 0, memo).mejorDesdeAqui;

        Console.WriteLine($"Racha máxima de {x} consecutivos: {mejor}");
    }

    // Devuelve:
    // - mejorDesdeAqui: mejor racha en [i..fin]
    // - rachaDesdeAqui: racha consecutiva de x que empieza exactamente en i
    static (int mejorDesdeAqui, int rachaDesdeAqui) MejorRachaRecDP(
        int[] a, int x, int i,
        Dictionary<int, (int mejorDesdeAqui, int rachaDesdeAqui)> memo)
    {
        if (i >= a.Length) return (0, 0);

        if (memo.TryGetValue(i, out var ya))
            return ya;

        var siguiente = MejorRachaRecDP(a, x, i + 1, memo);

        int rachaAqui = (a[i] == x) ? 1 + siguiente.rachaDesdeAqui : 0;
        int mejorAqui = (rachaAqui > siguiente.mejorDesdeAqui) ? rachaAqui : siguiente.mejorDesdeAqui;

        var res = (mejorAqui, rachaAqui);
        memo[i] = res;
        return res;
    }

    // -------- Entrada recursiva --------
    static void LeerArregloRec(int[] arr, int i)
    {
        if (i >= arr.Length) return;
        arr[i] = LeerEnteroRec($"Elemento {i + 1}: ");
        LeerArregloRec(arr, i + 1);
    }

    static int LeerEnteroRec(string mensaje)
    {
        Console.Write(mensaje);
        string t = Console.ReadLine() ?? "";
        if (TryParseIntRec(t, 0, false, 0, out int v)) return v;

        Console.WriteLine("Entrada inválida. Intenta de nuevo.");
        return LeerEnteroRec(mensaje);
    }

    static bool TryParseIntRec(string t, int i, bool neg, int acc, out int v)
    {
        if (i >= t.Length) { v = neg ? -acc : acc; return t.Length > 0; }
        char c = t[i];
        if (i == 0 && c == '-') return TryParseIntRec(t, i + 1, true, 0, out v);
        if (c < '0' || c > '9') { v = 0; return false; }
        return TryParseIntRec(t, i + 1, neg, acc * 10 + (c - '0'), out v);
    }
}



B) Iterativo con DP (tabulación)

using System;

class RachaMaximaX_IterativoDP
{
    static void Main()
    {
        Console.Write("Cantidad N: ");
        int n = int.Parse(Console.ReadLine() ?? "0");

        int[] a = new int[n];
        for (int i = 0; i < n; i++)
        {
            Console.Write($"Elemento {i + 1}: ");
            a[i] = int.Parse(Console.ReadLine() ?? "0");
        }

        Console.Write("Número x: ");
        int x = int.Parse(Console.ReadLine() ?? "0");

        // dpRacha[i] = racha consecutiva de x que termina en i
        int[] dpRacha = new int[n];
        int mejor = 0;

        for (int i = 0; i < n; i++)
        {
            if (a[i] == x)
                dpRacha[i] = (i == 0) ? 1 : dpRacha[i - 1] + 1;
            else
                dpRacha[i] = 0;

            if (dpRacha[i] > mejor) mejor = dpRacha[i];
        }

        Console.WriteLine($"Racha máxima de {x} consecutivos: {mejor}");
    }
}
