Contar cuántas rachas (grupos) de x consecutivos existen

A) Recursivo con DP (memoización) + entrada recursiva

using System;
using System.Collections.Generic;

class ContarRachasX_RecursivoDP
{
    static void Main()
    {
        int n = LeerEnteroRec("Cantidad N: ");
        int[] a = new int[n];
        LeerArregloRec(a, 0);

        int x = LeerEnteroRec("Valor x: ");

        // DP por índice: devuelve (rachasDesdeAqui, estoyEnRacha)
        var memo = new Dictionary<int, (int rachas, bool enRacha)>();
        int rachas = ContarRachasRecDP(a, x, 0, false, memo);

        Console.WriteLine($"Cantidad de rachas de {x}: {rachas}");
    }

    // estado: enRacha indica si venimos de un x justo antes
    // memo por (i, enRacha) usando clave string
    static int ContarRachasRecDP(int[] a, int x, int i, bool enRacha, Dictionary<int, (int rachas, bool enRacha)> memoBase)
    {
        // Para memo con estado, usamos dos diccionarios “aplanados”:
        // clave = i*2 + (enRacha?1:0)
        // guardamos SOLO rachas (enRacha no se guarda, se deduce del estado)
        return ContarRachasRecDP_Clave(a, x, i, enRacha, new Dictionary<int, int>());
    }

    static int ContarRachasRecDP_Clave(int[] a, int x, int i, bool enRacha, Dictionary<int, int> memo)
    {
        if (i >= a.Length) return 0;

        int clave = i * 2 + (enRacha ? 1 : 0);
        if (memo.TryGetValue(clave, out int ya))
            return ya;

        int res;

        if (a[i] == x)
        {
            // si no veníamos en racha, se inicia una nueva racha
            int suma = enRacha ? 0 : 1;
            res = suma + ContarRachasRecDP_Clave(a, x, i + 1, true, memo);
        }
        else
        {
            // se rompe racha
            res = ContarRachasRecDP_Clave(a, x, i + 1, false, memo);
        }

        memo[clave] = res;
        return res;
    }

    // -------- Entrada recursiva (sin ciclos) --------
    static void LeerArregloRec(int[] arr, int i)
    {
        if (i >= arr.Length) return;
        arr[i] = LeerEnteroRec($"Elemento {i + 1}: ");
        LeerArregloRec(arr, i + 1);
    }

    static int LeerEnteroRec(string mensaje)
    {
        Console.Write(mensaje);
        string t = Console.ReadLine() ?? "";
        if (TryParseIntRec(t, 0, false, 0, out int v)) return v;

        Console.WriteLine("Entrada inválida. Intenta de nuevo.");
        return LeerEnteroRec(mensaje);
    }

    static bool TryParseIntRec(string t, int i, bool neg, int acc, out int v)
    {
        if (i >= t.Length) { v = neg ? -acc : acc; return t.Length > 0; }
        char c = t[i];
        if (i == 0 && c == '-') return TryParseIntRec(t, i + 1, true, 0, out v);
        if (c < '0' || c > '9') { v = 0; return false; }
        return TryParseIntRec(t, i + 1, neg, acc * 10 + (c - '0'), out v);
    }
}



B) Iterativo con DP (tabulación)

using System;

class ContarRachasX_IterativoDP
{
    static void Main()
    {
        Console.Write("Cantidad N: ");
        int n = int.Parse(Console.ReadLine() ?? "0");

        int[] a = new int[n];
        for (int i = 0; i < n; i++)
        {
            Console.Write($"Elemento {i + 1}: ");
            a[i] = int.Parse(Console.ReadLine() ?? "0");
        }

        Console.Write("Valor x: ");
        int x = int.Parse(Console.ReadLine() ?? "0");

        // dpEnRacha[i] = true si en i estamos dentro de una racha de x
        // dpRachas[i] = cantidad de rachas en 0..i
        bool[] dpEnRacha = new bool[n];
        int[] dpRachas = new int[n];

        if (n > 0)
        {
            dpEnRacha[0] = (a[0] == x);
            dpRachas[0] = (a[0] == x) ? 1 : 0;
        }

        for (int i = 1; i < n; i++)
        {
            if (a[i] == x)
            {
                dpEnRacha[i] = true;
                // inicia una racha si antes NO estábamos en racha
                dpRachas[i] = dpRachas[i - 1] + (dpEnRacha[i - 1] ? 0 : 1);
            }
            else
            {
                dpEnRacha[i] = false;
                dpRachas[i] = dpRachas[i - 1];
            }
        }

        Console.WriteLine($"Cantidad de rachas de {x}: {(n == 0 ? 0 : dpRachas[n - 1])}");
    }
}
