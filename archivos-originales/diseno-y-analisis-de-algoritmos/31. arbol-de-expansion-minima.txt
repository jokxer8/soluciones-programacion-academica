Método por Arista: 


using System;
using System.Collections.Generic;

class Arista
{
    public int Origen;
    public int Destino;
    public int Peso;

    public Arista(int origen, int destino, int peso)
    {
        Origen = origen;
        Destino = destino;
        Peso = peso;
    }
}

class Program
{
    static void Main()
    {
        List<Arista> aristas = new List<Arista>();

        aristas.Add(new Arista(0, 1, 1));
        aristas.Add(new Arista(0, 2, 4));
        aristas.Add(new Arista(1, 2, 2));
        aristas.Add(new Arista(1, 3, 5));
        aristas.Add(new Arista(2, 3, 3));

        int cantidadVertices = 4;
        int menorPeso = int.MaxValue;
        List<Arista> mejorArbol = null;
        int numeroArbol = 1;

        for (int i = 0; i < aristas.Count - 2; i++)
        {
            for (int j = i + 1; j < aristas.Count - 1; j++)
            {
                for (int k = j + 1; k < aristas.Count; k++)
                {
                    List<Arista> posibleArbol = new List<Arista>();
                    posibleArbol.Add(aristas[i]);
                    posibleArbol.Add(aristas[j]);
                    posibleArbol.Add(aristas[k]);

                    if (EsArbolExpansion(posibleArbol, cantidadVertices))
                    {
                        int pesoTotal = CalcularPeso(posibleArbol);

                        Console.WriteLine("Arbol de expansion " + numeroArbol + ":");
                        MostrarArbol(posibleArbol, pesoTotal);

                        if (pesoTotal < menorPeso)
                        {
                            menorPeso = pesoTotal;
                            mejorArbol = new List<Arista>(posibleArbol);
                        }

                        numeroArbol++;
                    }
                }
            }
        }

        Console.WriteLine("MST (Arbol de Expansion Minima):");
        MostrarArbol(mejorArbol, menorPeso);
    }

    static bool EsArbolExpansion(List<Arista> arbol, int cantidadVertices)
    {
        if (arbol.Count != cantidadVertices - 1)
            return false;

        int[] componentes = new int[cantidadVertices];

        for (int i = 0; i < cantidadVertices; i++)
            componentes[i] = i;

        for (int i = 0; i < arbol.Count; i++)
            UnirComponentes(componentes, arbol[i].Origen, arbol[i].Destino);

        for (int i = 1; i < cantidadVertices; i++)
        {
            if (componentes[i] != componentes[0])
                return false;
        }

        return true;
    }

    static void UnirComponentes(int[] componentes, int origen, int destino)
    {
        int valorOrigen = componentes[origen];
        int valorDestino = componentes[destino];

        if (valorOrigen == valorDestino)
            return;

        for (int i = 0; i < componentes.Length; i++)
        {
            if (componentes[i] == valorDestino)
                componentes[i] = valorOrigen;
        }
    }

    static int CalcularPeso(List<Arista> arbol)
    {
        int suma = 0;

        for (int i = 0; i < arbol.Count; i++)
            suma += arbol[i].Peso;

        return suma;
    }

    static void MostrarArbol(List<Arista> arbol, int pesoTotal)
    {
        for (int i = 0; i < arbol.Count; i++)
        {
            Console.WriteLine("(" + arbol[i].Origen + ", " + arbol[i].Destino + ") = " + arbol[i].Peso);
        }

        Console.WriteLine("Peso total = " + pesoTotal);
        Console.WriteLine();
    }
}


Metodo por NODO 

using System;
using System.Collections.Generic;

class Nodo
{
    public int Valor;
    public int Clase;

    public Nodo(int valor)
    {
        Valor = valor;
        Clase = valor; // al inicio cada nodo pertenece a su propia clase
    }
}

class Arista
{
    public Nodo Origen;
    public Nodo Destino;
    public int Peso;

    public Arista(Nodo origen, Nodo destino, int peso)
    {
        Origen = origen;
        Destino = destino;
        Peso = peso;
    }
}

class Program
{
    static void Main()
    {
        // Crear nodos
        Nodo nodo0 = new Nodo(0);
        Nodo nodo1 = new Nodo(1);
        Nodo nodo2 = new Nodo(2);
        Nodo nodo3 = new Nodo(3);

        List<Nodo> nodos = new List<Nodo>();
        nodos.Add(nodo0);
        nodos.Add(nodo1);
        nodos.Add(nodo2);
        nodos.Add(nodo3);

        // Crear aristas ya ordenadas de menor a mayor
        List<Arista> aristas = new List<Arista>();
        aristas.Add(new Arista(nodo0, nodo1, 1));
        aristas.Add(new Arista(nodo1, nodo2, 2));
        aristas.Add(new Arista(nodo2, nodo3, 3));
        aristas.Add(new Arista(nodo0, nodo2, 4));
        aristas.Add(new Arista(nodo1, nodo3, 5));

        List<Arista> arbolExpansionMinima = new List<Arista>();
        int pesoTotal = 0;

        for (int i = 0; i < aristas.Count; i++)
        {
            Nodo origen = aristas[i].Origen;
            Nodo destino = aristas[i].Destino;

            if (origen.Clase != destino.Clase)
            {
                arbolExpansionMinima.Add(aristas[i]);
                pesoTotal += aristas[i].Peso;

                UnirClases(nodos, origen.Clase, destino.Clase);
            }
        }

        Console.WriteLine("Arbol de Expansion Minima:");
        for (int i = 0; i < arbolExpansionMinima.Count; i++)
        {
            Console.WriteLine(
                "(" + arbolExpansionMinima[i].Origen.Valor + ", " +
                arbolExpansionMinima[i].Destino.Valor + ") -> " +
                arbolExpansionMinima[i].Peso
            );
        }

        Console.WriteLine("Peso total: " + pesoTotal);
    }

    static void UnirClases(List<Nodo> nodos, int clase1, int clase2)
    {
        for (int i = 0; i < nodos.Count; i++)
        {
            if (nodos[i].Clase == clase2)
            {
                nodos[i].Clase = clase1;
            }
        }
    }
}
