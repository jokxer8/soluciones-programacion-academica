Recorrido recursivo de un arreglo dado

A) Recursivo con DP (memoización) + entrada recursiva

using System;
using System.Collections.Generic;

class RecorrerArreglo_RecursivoDP
{
    static void Main()
    {
        int n = LeerEnteroRec("Cantidad N: ");
        int[] arreglo = new int[n];

        LeerArregloRec(arreglo, 0);

        Console.WriteLine("\nRecorrido del arreglo:");
        var memoVisitado = new Dictionary<int, bool>();
        RecorrerRecDP(arreglo, 0, memoVisitado);
        Console.WriteLine();
    }

    static void RecorrerRecDP(int[] arreglo, int i, Dictionary<int, bool> memoVisitado)
    {
        if (i >= arreglo.Length) return;

        if (memoVisitado.ContainsKey(i)) return;
        memoVisitado[i] = true;

        Console.Write(arreglo[i] + " ");
        RecorrerRecDP(arreglo, i + 1, memoVisitado);
    }

    // -------- Entrada recursiva (sin ciclos) --------
    static void LeerArregloRec(int[] arreglo, int i)
    {
        if (i >= arreglo.Length) return;
        arreglo[i] = LeerEnteroRec($"Elemento {i + 1}: ");
        LeerArregloRec(arreglo, i + 1);
    }

    static int LeerEnteroRec(string mensaje)
    {
        Console.Write(mensaje);
        string texto = Console.ReadLine() ?? "";

        if (TryParseIntRec(texto, 0, false, 0, out int valor))
            return valor;

        Console.WriteLine("Entrada inválida. Intenta de nuevo.");
        return LeerEnteroRec(mensaje);
    }

    static bool TryParseIntRec(string t, int i, bool neg, int acc, out int v)
    {
        if (i >= t.Length) { v = neg ? -acc : acc; return t.Length > 0; }
        char c = t[i];
        if (i == 0 && c == '-') return TryParseIntRec(t, i + 1, true, 0, out v);
        if (c < '0' || c > '9') { v = 0; return false; }
        return TryParseIntRec(t, i + 1, neg, acc * 10 + (c - '0'), out v);
    }
}



B) Iterativo con DP (tabulación)

using System;

class RecorrerArreglo_IterativoDP
{
    static void Main()
    {
        Console.Write("Cantidad N: ");
        int n = int.Parse(Console.ReadLine() ?? "0");

        int[] arreglo = new int[n];
        for (int i = 0; i < n; i++)
        {
            Console.Write($"Elemento {i + 1}: ");
            arreglo[i] = int.Parse(Console.ReadLine() ?? "0");
        }

        // DP: dp[i] "representa" el elemento i (tabulación 1 a 1)
        int[] dp = new int[n];
        for (int i = 0; i < n; i++)
            dp[i] = arreglo[i];

        Console.WriteLine("\nRecorrido del arreglo:");
        for (int i = 0; i < n; i++)
            Console.Write(dp[i] + " ");
        Console.WriteLine();
    }
}
