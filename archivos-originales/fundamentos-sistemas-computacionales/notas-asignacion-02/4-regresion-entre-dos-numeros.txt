4. Escriba un programa en C y ensamblador que pida al usuario dos números enteros positivos m y n, y luego imprima los números comprendidos entre n y m, incluidos ambos, de manera regresiva.

C:

#include <stdio.h>

int main(void) {
    int m, n, temp;

    printf("Digite un numero entero positivo: ");
    scanf("%d", &m);

    printf("Digite otro numero entero positivo: ");
    scanf("%d", &n);

    if (m < n) {
        temp = m;
        m = n;
        n = temp;
    }

    printf("Los numeros entre en ese rango son:\n");

    for (int i = m; i >= n; i--) {
        printf("%d", i);

        if (i > n) {
            printf(", ");
        }
    }

    printf("\n");

    return 0;
}


Ensamblador FASM x86-64:


format ELF64 executable
entry inicio

segment readable executable

inicio:
    mov rsi, pedir_m
    mov rdx, pedir_m_1
    call imprimir
    call leer_int
    mov r12, rax
    call salto_linea

    mov rsi, pedir_n
    mov rdx, pedir_n_1
    call imprimir
    call leer_int
    mov r13, rax
    call salto_linea

    cmp r12, r13
    jge continuar

    mov r14, r12
    mov r12, r13
    mov r13, r14

continuar:
    mov rsi, titulo
    mov rdx, titulo_1
    call imprimir

    mov r14, r12

ciclo:
    cmp r14, r13
    jl fin

    mov rax, r14
    call imprimir_int

    cmp r14, r13
    je sin_coma

    mov rsi, coma
    mov rdx, coma_1
    call imprimir

sin_coma:
    dec r14
    jmp ciclo

fin:
    call salto_linea
    call salir

imprimir:
    mov rax, 1
    mov rdi, 1
    syscall
    ret

leer_int:
    mov rax, 0
    mov rdi, 0
    mov rsi, memoria
    mov rdx, 64
    syscall

    mov rcx, rax
    mov rsi, memoria
    xor rax, rax
    xor r8, r8

buscar:
    cmp rcx, 0
    je fin_leer

    mov bl, [rsi]

    cmp bl, '-'
    je negativo

    cmp bl, '0'
    jb avanzar
    cmp bl, '9'
    jbe leer_digitos

avanzar:
    inc rsi
    dec rcx
    jmp buscar

negativo:
    mov r8, 1
    inc rsi
    dec rcx

leer_digitos:
    cmp rcx, 0
    je aplicar_signo

    mov bl, [rsi]
    cmp bl, '0'
    jb aplicar_signo
    cmp bl, '9'
    ja aplicar_signo

    imul rax, rax, 10
    movzx rbx, bl
    sub rbx, '0'
    add rax, rbx

    inc rsi
    dec rcx
    jmp leer_digitos

aplicar_signo:
    cmp r8, 1
    jne fin_leer
    neg rax

fin_leer:
    ret

imprimir_int:
    mov rsi, numero_memoria + 32
    mov rbx, 10
    xor r9, r9

    cmp rax, 0
    jne verificar_signo

    dec rsi
    mov byte [rsi], '0'
    mov rdx, 1
    jmp imprimir_numero

verificar_signo:
    cmp rax, 0
    jge convertir

    neg rax
    mov r9, 1

convertir:
    xor rdx, rdx
    div rbx

    add dl, '0'
    dec rsi
    mov [rsi], dl

    test rax, rax
    jne convertir

    cmp r9, 1
    jne calcular_longitud

    dec rsi
    mov byte [rsi], '-'

calcular_longitud:
    mov rdx, numero_memoria + 32
    sub rdx, rsi

imprimir_numero:
    call imprimir
    ret

salto_linea:
    mov rsi, nueva_linea
    mov rdx, 1
    call imprimir
    ret

salir:
    mov rax, 60
    xor rdi, rdi
    syscall

segment readable writeable

memoria rb 64
numero_memoria rb 32
nueva_linea db 10

pedir_m db 'Digite un numero entero positivo: '
pedir_m_1 = $ - pedir_m

pedir_n db 'Digite otro numero entero positivo: '
pedir_n_1 = $ - pedir_n

titulo db 'Los numeros entre en ese rango son:', 10
titulo_1 = $ - titulo

coma db ', '
coma_1 = $ - coma
