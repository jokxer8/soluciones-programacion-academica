8.2 Se desea un programa en lenguaje C y ensamblador que pida al usuario un numero entre 1 y 20 (enteros positivos) e imprima un patron de asteriscos como se muestran en los siguientes ejemplos. Note que en todos los ejemplos se representa que el usuario digito el numero 5.

Digite un # entre 1 - 20: 5

*
**
***
****
*****


C:

#include <stdio.h>

int main(void) {
    int n;

    printf("Digite un # entre 1 - 20: ");
    scanf("%d", &n);

    if (n < 1 || n > 20) {
        printf("Numero fuera de rango.\n");
        return 0;
    }

    for (int fila = 1; fila <= n; fila++) {
        for (int estrella = 1; estrella <= fila; estrella++) {
            printf("*");
        }

        printf("\n");
    }

    return 0;
}



Ensamblador FASM x86-64:

format ELF64 executable
entry inicio

segment readable executable

inicio:
    mov rsi, pedir_n
    mov rdx, pedir_n_1
    call imprimir
    call leer_int
    mov r12, rax
    call nueva_linea

    cmp r12, 1
    jl fuera_rango
    cmp r12, 20
    jg fuera_rango

    mov r13, 1

filas:
    cmp r13, r12
    jg fin

    mov r14, 1

estrellas:
    cmp r14, r13
    jg nueva_linea

    mov al, '*'
    call imprimir_caracter

    inc r14
    jmp estrellas

nueva_linea:
    mov al, 10
    call imprimir_caracter

    inc r13
    jmp filas

fuera_rango:
    mov rsi, error
    mov rdx, error_1
    call imprimir

fin:
    call salir

imprimir:
    mov rax, 1
    mov rdi, 1
    syscall
    ret

imprimir_caracter:
    mov [caracter], al
    mov rsi, caracter
    mov rdx, 1
    call imprimir
    ret

leer_int:
    mov rax, 0
    mov rdi, 0
    mov rsi, memoria
    mov rdx, 64
    syscall

    mov rcx, rax
    mov rsi, memoria
    xor rax, rax

leer:
    cmp rcx, 0
    je fin_leer

    mov bl, [rsi]
    cmp bl, '0'
    jb avanzar
    cmp bl, '9'
    ja fin_leer

    imul rax, rax, 10
    movzx rbx, bl
    sub rbx, '0'
    add rax, rbx

avanzar:
    inc rsi
    dec rcx
    jmp leer

fin_leer:
    ret

salir:
    mov rax, 60
    xor rdi, rdi
    syscall

segment readable writeable

memoria rb 64
caracter db 0

pedir_n db 'Digite un # entre 1 - 20: '
pedir_n_1 = $ - pedir_n

error db 'Numero fuera de rango.', 10
error_1 = $ - error
