Codigo del problema:

using System;

namespace SumaRecursivaArreglo
{
    // Clase base
    public abstract class ProcesadorArregloBase
    {
        // Método abstracto para la suma recursiva
        public abstract int SumaRecursiva(int[] arreglo, int longitud);
    }

    // Clase derivada que implementa el método abstracto
    public class ProcesadorArreglo : ProcesadorArregloBase
    {
        // Implementación del método recursivo para sumar los elementos del arreglo
        public override int SumaRecursiva(int[] arreglo, int longitud)
        {
            // Caso base: si el arreglo está vacío, la suma es 0
            if (longitud == 0)
            {
                return 0;
            }
            // Caso recursivo: suma el último elemento y llama recursivamente con el resto del arreglo
            return arreglo[longitud - 1] + SumaRecursiva(arreglo, longitud - 1);
        }

        // Método para leer y validar los datos desde la consola
        public void LeerYProcesarArreglo()
        {
            int longitud = 0;
            while (true)
            {
                Console.Write("Ingrese la cantidad de elementos del arreglo: ");
                string entradaLongitud = Console.ReadLine();
                if (int.TryParse(entradaLongitud, out longitud) && longitud > 0)
                {
                    break;
                }
                Console.WriteLine("Entrada inválida. Debe ingresar un número entero positivo.");
            }

            int[] arreglo = new int[longitud];
            for (int i = 0; i < longitud; i++)
            {
                while (true)
                {
                    Console.Write($"Ingrese el elemento {i + 1}: ");
                    string entradaElemento = Console.ReadLine();
                    if (int.TryParse(entradaElemento, out arreglo[i]))
                    {
                        break;
                    }
                    Console.WriteLine("Entrada inválida. Debe ingresar un número entero.");
                }
            }

            int suma = SumaRecursiva(arreglo, longitud);
            Console.WriteLine($"La suma de todos los elementos del arreglo es: {suma}");
        }
    }

    class Programa
    {
        static void Main(string[] args)
        {
            ProcesadorArreglo procesadorArreglo = new ProcesadorArreglo();
            procesadorArreglo.LeerYProcesarArreglo();
        }
    }
}



Explicación del Código:
1. Clase Base ProcesadorArregloBase:

Define un método abstracto SumaRecursiva que toma un arreglo y su longitud.

2.Clase Derivada ProcesadorArreglo:

Implementa el método SumaRecursiva para calcular la suma de los elementos de manera recursiva.
Incluye el método LeerYProcesarArreglo que se encarga de leer y validar los datos ingresados por el usuario.

3. Método SumaRecursiva:

Caso base: Si la longitud del arreglo es 0, retorna 0.
Caso recursivo: Suma el último elemento del arreglo y llama recursivamente al método con la longitud disminuida en 1.

4.Método LeerYProcesarArreglo:

Lee y valida la cantidad de elementos del arreglo.
Lee y valida cada elemento del arreglo.
Calcula y muestra la suma de todos los elementos usando el método recursivo.

5. Ejecución del Programa:
El programa solicita al usuario la cantidad de elementos del arreglo y luego cada elemento del mismo.
Calcula recursivamente la suma de los elementos y la muestra en pantalla.
Este programa utiliza clases, métodos regulares, polimorfismo dinámico, y validación de entrada de datos, cumpliendo con los requisitos especificados.
