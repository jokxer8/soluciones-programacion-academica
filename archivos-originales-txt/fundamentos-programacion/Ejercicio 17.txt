you are given two positive integers a and b. In one move you can increase a by 1 (replace a
With a + 1). Your task is to find the minimum number of moves you need to do in order to
make a divisible by b. It is possible, that you have to make 0 moves, as a is already divisible
by b You have to answer t independent test cases.
Input
The first line of the input contains one integer t (l <= t <= 10^4) — the number of test cases.
Then t test cases follow.
The only line of the test case contains two integers a and b (I <= a, b <= 10^9).

Output
For each test case print the answer — the minimum number of moves you need to do in order
to make a divisible by b.



using System;

class Program
{
    static int CalculateMoves(int a, int b)
    {
        if (a % b == 0)
        {
            return 0;
        }
        else
        {
            int remainder = a % b;
            return b - remainder;
        }
    }
    static void Main()
    {
        int t = int.Parse(Console.ReadLine());

        for (int i = 0; i < t; i++)
        {
            string[] input = Console.ReadLine().Split();
            int a = int.Parse(input[0]);
            int b = int.Parse(input[1]);

            int moves = CalculateMoves(a, b);
            Console.WriteLine(moves);
        }
        Console.ReadLine();
    }
}
