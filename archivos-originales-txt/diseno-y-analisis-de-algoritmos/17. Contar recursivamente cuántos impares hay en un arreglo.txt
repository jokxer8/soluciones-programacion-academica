Contar recursivamente cuántos impares hay en un arreglo

A) Recursivo con DP + entrada recursiva

using System;
using System.Collections.Generic;

class ContarImpares_RecursivoDP
{
    static void Main()
    {
        int n = LeerEnteroRec("Cantidad N: ");
        int[] arreglo = new int[n];
        LeerArregloRec(arreglo, 0);

        var memo = new Dictionary<int, int>();
        int impares = ContarImparesRecDP(arreglo, 0, memo);

        Console.WriteLine("Cantidad de impares: " + impares);
    }

    static int ContarImparesRecDP(int[] arreglo, int i, Dictionary<int, int> memo)
    {
        if (i >= arreglo.Length) return 0;

        if (memo.TryGetValue(i, out int ya))
            return ya;

        int actual = (arreglo[i] % 2 != 0) ? 1 : 0;
        int res = actual + ContarImparesRecDP(arreglo, i + 1, memo);

        memo[i] = res;
        return res;
    }

    // -------- Entrada recursiva (sin ciclos) --------
    static void LeerArregloRec(int[] arr, int i)
    {
        if (i >= arr.Length) return;
        arr[i] = LeerEnteroRec($"Elemento {i + 1}: ");
        LeerArregloRec(arr, i + 1);
    }

    static int LeerEnteroRec(string mensaje)
    {
        Console.Write(mensaje);
        string t = Console.ReadLine() ?? "";
        if (TryParseIntRec(t, 0, false, 0, out int v)) return v;

        Console.WriteLine("Entrada inválida. Intenta de nuevo.");
        return LeerEnteroRec(mensaje);
    }

    static bool TryParseIntRec(string t, int i, bool neg, int acc, out int v)
    {
        if (i >= t.Length) { v = neg ? -acc : acc; return t.Length > 0; }
        char c = t[i];
        if (i == 0 && c == '-') return TryParseIntRec(t, i + 1, true, 0, out v);
        if (c < '0' || c > '9') { v = 0; return false; }
        return TryParseIntRec(t, i + 1, neg, acc * 10 + (c - '0'), out v);
    }
}



B) Iterativo con DP (tabulación)

using System;

class ContarImpares_IterativoDP
{
    static void Main()
    {
        Console.Write("Cantidad N: ");
        int n = int.Parse(Console.ReadLine() ?? "0");

        int[] arreglo = new int[n];
        for (int i = 0; i < n; i++)
        {
            Console.Write($"Elemento {i + 1}: ");
            arreglo[i] = int.Parse(Console.ReadLine() ?? "0");
        }

        // dp[i] = cantidad de impares en arreglo[0..i]
        int[] dp = new int[n];
        if (n > 0)
            dp[0] = (arreglo[0] % 2 != 0) ? 1 : 0;

        for (int i = 1; i < n; i++)
            dp[i] = dp[i - 1] + ((arreglo[i] % 2 != 0) ? 1 : 0);

        Console.WriteLine("Cantidad de impares: " + (n == 0 ? 0 : dp[n - 1]));
    }
}
