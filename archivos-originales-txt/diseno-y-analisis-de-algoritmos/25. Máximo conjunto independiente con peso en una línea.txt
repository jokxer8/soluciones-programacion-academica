Máximo conjunto independiente con peso en una línea

(“no tomar adyacentes”, tipo House Robber)
Ejemplo: [3,1,4,1,5,9,2,6] → 22

A) Recursivo con DP + entrada recursiva

using System;
using System.Collections.Generic;

class MISLinea_RecursivoDP
{
    static void Main()
    {
        int n = LeerEnteroRec("Cantidad N: ");
        int[] pesos = new int[n];
        LeerArregloRec(pesos, 0);

        var memo = new Dictionary<int, long>();
        long mejor = MejorRecDP(pesos, 0, memo);

        Console.WriteLine("Máximo peso sin adyacentes: " + mejor);
    }

    // dp[i] = mejor desde i hasta el final
    static long MejorRecDP(int[] a, int i, Dictionary<int, long> memo)
    {
        if (i >= a.Length) return 0;

        if (memo.TryGetValue(i, out long ya))
            return ya;

        long tomar = a[i] + MejorRecDP(a, i + 2, memo);
        long saltar = MejorRecDP(a, i + 1, memo);

        long res = (tomar > saltar) ? tomar : saltar;
        memo[i] = res;
        return res;
    }

    // -------- Entrada recursiva (sin ciclos) --------
    static void LeerArregloRec(int[] arr, int i)
    {
        if (i >= arr.Length) return;
        arr[i] = LeerEnteroRec($"Peso {i + 1}: ");
        LeerArregloRec(arr, i + 1);
    }

    static int LeerEnteroRec(string mensaje)
    {
        Console.Write(mensaje);
        string t = Console.ReadLine() ?? "";
        if (TryParseIntRec(t, 0, false, 0, out int v)) return v;

        Console.WriteLine("Entrada inválida. Intenta de nuevo.");
        return LeerEnteroRec(mensaje);
    }

    static bool TryParseIntRec(string t, int i, bool neg, int acc, out int v)
    {
        if (i >= t.Length) { v = neg ? -acc : acc; return t.Length > 0; }
        char c = t[i];
        if (i == 0 && c == '-') return TryParseIntRec(t, i + 1, true, 0, out v);
        if (c < '0' || c > '9') { v = 0; return false; }
        return TryParseIntRec(t, i + 1, neg, acc * 10 + (c - '0'), out v);
    }
}



B) Iterativo con DP (tabulación)

using System;

class MISLinea_IterativoDP
{
    static void Main()
    {
        Console.Write("Cantidad N: ");
        int n = int.Parse(Console.ReadLine() ?? "0");

        int[] a = new int[n];
        for (int i = 0; i < n; i++)
        {
            Console.Write($"Peso {i + 1}: ");
            a[i] = int.Parse(Console.ReadLine() ?? "0");
        }

        // dp[i] = mejor usando a[0..i]
        if (n == 0) { Console.WriteLine("Máximo peso sin adyacentes: 0"); return; }

        long[] dp = new long[n];
        dp[0] = Math.Max(0, a[0]);

        if (n >= 2)
            dp[1] = Math.Max(dp[0], a[1]);

        for (int i = 2; i < n; i++)
            dp[i] = Math.Max(dp[i - 1], dp[i - 2] + a[i]);

        Console.WriteLine("Máximo peso sin adyacentes: " + dp[n - 1]);
    }
}
